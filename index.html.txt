async function leerDatos() {
    try {
        log('GET ' + API_URL);
        setDiag('Leyendo del servidor...');
        const res = await fetch(API_URL + '?t=' + Date.now());
        if (!res.ok) { setDiag('Error GET: HTTP ' + res.status, true); throw new Error('HTTP ' + res.status); }
        const data = await res.json();
        log('Datos:', data);
        setDiag('GET OK');
        return data || {};
    } catch (e) {
        console.error('GET falló:', e);
        setDiag('GET falló: ' + e.message, true);
        return null;
    }
}

async function escribirDatos(datos) {
    try {
        log('PUT ' + API_URL);
        setDiag('Escribiendo en servidor...');
        const res = await fetch(API_URL, {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(datos)
        });
        if (!res.ok) {
            const errTxt = await res.text();
            console.error('Error servidor:', res.status, errTxt);
            setDiag('Error PUT: HTTP ' + res.status, true);
            throw new Error('HTTP ' + res.status);
        }
        log('Guardado OK');
        setDiag('PUT OK');
        return true;
    } catch (e) {
        console.error('PUT falló:', e);
        setDiag('PUT falló: ' + e.message, true);
        return false;
    }
}

function hashPassword(pass) {
    let hash = 0;
    for (let i = 0; i < pass.length; i++) {
        const char = pass.charCodeAt(i);
        hash = ((hash << 5) - hash) + char;
        hash = hash & hash;
    }
    return 'h_' + Math.abs(hash).toString(36);
}

async function cargarDatosDelServidor() {
    mostrarSync('Cargando...', 'syncing');
    const datos = await leerDatos();

    if (datos && typeof datos === 'object') {
        let modificado = false;

        if (datos.usuarios && Object.keys(datos.usuarios).length > 0) {
            usuariosCache = datos.usuarios;
        } else {
            usuariosCache = { admin: { username: 'admin', nombre: 'Administrador', pass: hashPassword('admin123'), rol: 'admin', activo: true, tema: 'futurista', creado: new Date().toISOString() } };
            modificado = true;
        }

        if (datos.plantillas && Object.keys(datos.plantillas).length > 0) {
            plantillasCache = datos.plantillas;
        } else {
            plantillasCache = plantillasPorDefecto();
            modificado = true;
        }

        if (modificado) {
            await escribirDatos({ plantillas: plantillasCache, usuarios: usuariosCache, version: 2 });
        }

        localStorage.setItem(CACHE_KEY, JSON.stringify({ usuarios: usuariosCache, plantillas: plantillasCache }));
        mostrarSync('Datos cargados', 'success');
    } else {
        const cache = localStorage.getItem(CACHE_KEY);
        if (cache) {
            try {
                const c = JSON.parse(cache);
                usuariosCache = c.usuarios || {};
                plantillasCache = c.plantillas || plantillasPorDefecto();
                mostrarSync('Usando caché local', 'error');
            } catch (e) {
                usuariosCache = {};
                plantillasCache = plantillasPorDefecto();
            }

    try {

}
